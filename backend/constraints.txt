# references it with -c, so pip, CI, Docker and Render all honour it.
#
# Regenerate after changing requirements*.txt, from a fresh venv:
#   python -m venv /tmp/lock && /tmp/lock/bin/pip install -r requirements-dev.txt
#   /tmp/lock/bin/pip freeze --all --exclude pip --exclude setuptools  (body below)
#
# colorama is Windows-only (pytest's console colours); a constraint on a package
# that is never installed is inert.
